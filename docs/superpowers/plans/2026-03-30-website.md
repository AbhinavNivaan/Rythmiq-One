# Rythmiq Website Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build and deploy a dark, animated, single-page marketing website for Rythmiq One at rythmiq.in targeting investors, accelerators, and B2B partners.

**Architecture:** Next.js (static export) with Framer Motion for animations. One page composed of section components. Contact form via Formspree. Deployed to Firebase Hosting via `next build` → `out/` → `firebase deploy`.

**Tech Stack:** Next.js 14, TypeScript, Framer Motion, Formspree, Firebase Hosting, Google Analytics 4

**Spec:** `docs/superpowers/specs/2026-03-30-website-visual-redesign.md`

interface SectionPillProps {

const accentStyles = {
  green: {
    border: 'rgba(52, 199, 89, 0.4)',
    color: 'rgba(52, 199, 89, 0.9)',
    dot: '#34C759',
    dotGlow: 'rgba(52, 199, 89, 0.8)',
  },
  red: {
    border: 'rgba(255, 59, 48, 0.4)',
    color: 'rgba(255, 59, 48, 0.9)',
    dot: '#FF3B30',
    dotGlow: 'rgba(255, 59, 48, 0.8)',
  },
  blue: {
    border: 'rgba(33, 150, 243, 0.4)',
    color: 'rgba(33, 150, 243, 0.9)',
    dot: '#2196f3',
    dotGlow: 'rgba(33, 150, 243, 0.9)',
  },
  white: {
    border: 'rgba(255, 255, 255, 0.14)',
    color: 'rgba(255, 255, 255, 0.45)',
    dot: null,
    dotGlow: null,
  },
}

export default function SectionPill({ label, accent = 'green' }: SectionPillProps) {
  const s = accentStyles[accent]
  return (
    <div style={{
      display: 'inline-flex',
      alignItems: 'center',
      gap: 8,
      border: `1px solid ${s.border}`,
      borderRadius: 100,
      padding: '6px 14px',
      fontSize: 11,
      fontWeight: 600,
      letterSpacing: '0.1em',
      textTransform: 'uppercase',
      color: s.color,
      marginBottom: 28,
      width: 'fit-content',
    }}>
      {s.dot && (
        <span style={{
          width: 6, height: 6,
          borderRadius: '50%',
          background: s.dot,
          boxShadow: `0 0 8px ${s.dotGlow}`,
          display: 'inline-block',
          animation: 'pill-dot-pulse 2s ease-in-out infinite',
        }} />
      )}
      {label}
      <style>{`
        @keyframes pill-dot-pulse {
          0%, 100% { opacity: 1; }
          50% { opacity: 0.35; }
        }
      `}</style>
    </div>
  )
}

  x: number; y: number
  tx: number; ty: number
  progress: number
  speed: number
  size: number
  delay: number
  active: boolean
}

export default function ParticleStream({
  fromRef, toRef, containerRef,
  color, glowColor, count = 20, active = true,
}: ParticleStreamProps) {
  const canvasRef = useRef<HTMLCanvasElement>(null)
  const animRef   = useRef<number>(0)
  const particles = useRef<Particle[]>([])

  function getRect(el: HTMLElement) {
    const cr = el.getBoundingClientRect()
    const br = containerRef.current!.getBoundingClientRect()
    return {
      top: cr.top - br.top, bottom: cr.bottom - br.top,
      left: cr.left - br.left, right: cr.right - br.left,
      width: cr.width, height: cr.height,
    }
  }

  function makeParticle(i: number): Particle {
    return {
      x: 0, y: 0, tx: 0, ty: 0,
      progress: 0,
      speed: 0.004 + Math.random() * 0.005,
      size: 2 + Math.random() * 2,
      delay: i * 20,
      active: false,
    }
  }

  function resetParticle(p: Particle) {
    if (!fromRef.current || !toRef.current || !containerRef.current) return
    const from = getRect(fromRef.current)
    const to   = getRect(toRef.current)
    p.x  = from.right - 8
    p.y  = from.top + 60 + Math.random() * Math.max(0, from.height - 120)
    p.tx = to.left + 8
    p.ty = to.top  + 60 + Math.random() * Math.max(0, to.height - 120)
    p.progress = 0
    p.speed    = 0.004 + Math.random() * 0.005
    p.size     = 2 + Math.random() * 2
    p.delay    = 0
    p.active   = false
  }

  useEffect(() => {
    particles.current = Array.from({ length: count }, (_, i) => makeParticle(i))

    const canvas = canvasRef.current
    if (!canvas) return
    const ctx = canvas.getContext('2d')!

    function resize() {
      const cont = containerRef.current
      if (!cont || !canvas) return
      canvas.width  = cont.offsetWidth
      canvas.height = cont.offsetHeight
    }
    resize()
    window.addEventListener('resize', resize)

    function draw() {
      if (!active) { animRef.current = requestAnimationFrame(draw); return }
      ctx.clearRect(0, 0, canvas!.width, canvas!.height)

      particles.current.forEach(p => {
        if (p.delay > 0) { p.delay--; return }
        p.active = true
        p.progress += p.speed
        if (p.progress >= 1) { resetParticle(p); return }

        const t    = p.progress
        const ease = t < 0.5 ? 4*t*t*t : 1 - Math.pow(-2*t+2, 3) / 2
        const mx   = (p.x + p.tx) / 2
        const my   = (p.y + p.ty) / 2 - 28

        const cx = (1-ease)*(1-ease)*p.x + 2*(1-ease)*ease*mx + ease*ease*p.tx
        const cy = (1-ease)*(1-ease)*p.y + 2*(1-ease)*ease*my + ease*ease*p.ty
        const alpha = Math.sin(Math.PI * ease)

        ctx.save()
        ctx.globalAlpha = alpha * 0.92
        ctx.shadowBlur  = 8
        ctx.shadowColor = glowColor
        ctx.fillStyle   = color
        ctx.beginPath()
        ctx.arc(cx, cy, p.size, 0, Math.PI * 2)
        ctx.fill()
        ctx.restore()

        // Trail
        const te  = Math.max(0, ease - 0.05)
        const tcx = (1-te)*(1-te)*p.x + 2*(1-te)*te*mx + te*te*p.tx
        const tcy = (1-te)*(1-te)*p.y + 2*(1-te)*te*my + te*te*p.ty
        ctx.save()
        ctx.globalAlpha = alpha * 0.25
        ctx.shadowBlur  = 4
        ctx.shadowColor = glowColor
        ctx.fillStyle   = color
        ctx.beginPath()
        ctx.arc(tcx, tcy, p.size * 0.55, 0, Math.PI * 2)
        ctx.fill()
        ctx.restore()
      })

      animRef.current = requestAnimationFrame(draw)
    }

    setTimeout(() => { resize(); draw() }, 300)
    return () => {
      cancelAnimationFrame(animRef.current)
      window.removeEventListener('resize', resize)
    }
  }, [active])

  return (
    <canvas
      ref={canvasRef}
      style={{
        position: 'absolute', inset: 0,
        width: '100%', height: '100%',
        pointerEvents: 'none', zIndex: 10,
      }}
    />
  )
}
